# 프로젝트 자동 세팅 가이드
> 새 프로젝트 시작 시 1분 만에 전체 환경 구성

---

## 방법 1: 셸 스크립트 (가장 빠름) — 터미널에서 직접 실행

### 1단계: 초기화 스크립트 생성 (1회만)

```bash
cat > ~/.claude/init-project.sh << 'INITEOF'
#!/bin/bash
# ═══════════════════════════════════════════
# Claude Code 프로젝트 초기화 스크립트
# 사용법: ~/.claude/init-project.sh [프로젝트타입]
# 프로젝트타입: nextjs (기본) | n8n | docs
# ═══════════════════════════════════════════

PROJECT_TYPE="${1:-nextjs}"
PROJECT_NAME=$(basename $(pwd))

echo "🚀 프로젝트 초기화: $PROJECT_NAME (타입: $PROJECT_TYPE)"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

# ── 1. .claude/ 폴더 구조 ──
mkdir -p .claude/agents
mkdir -p .claude/rules
mkdir -p .claude/skills
mkdir -p docs

# ── 2. 프로젝트별 settings.json ──
if [ "$PROJECT_TYPE" = "nextjs" ]; then
cat > .claude/settings.json << 'SEOF'
{
  "hooks": {
    "PostToolUse": [
      {
        "matcher": "Write|Edit",
        "hooks": [
          {
            "type": "command",
            "command": "case \"$CLAUDE_FILE_PATH\" in *.ts|*.tsx) npx tsc --noEmit --pretty 2>&1 | head -20 ;; esac; exit 0",
            "timeout": 30000
          }
        ]
      },
      {
        "matcher": "Write|Edit",
        "hooks": [
          {
            "type": "command",
            "command": "case \"$CLAUDE_FILE_PATH\" in *.ts|*.tsx) npx eslint --quiet --max-warnings 0 \"$CLAUDE_FILE_PATH\" 2>&1 | head -10 ;; esac; exit 0",
            "timeout": 15000
          }
        ]
      }
    ]
  }
}
SEOF
echo "  ✓ .claude/settings.json (Next.js + tsc + eslint)"

elif [ "$PROJECT_TYPE" = "n8n" ]; then
cat > .claude/settings.json << 'SEOF'
{
  "hooks": {}
}
SEOF
echo "  ✓ .claude/settings.json (n8n — Hook 최소화)"

else
cat > .claude/settings.json << 'SEOF'
{
  "hooks": {}
}
SEOF
echo "  ✓ .claude/settings.json (문서 — Hook 없음)"
fi

# ── 3. 프로젝트 CLAUDE.md ──
if [ ! -f CLAUDE.md ]; then
cat > CLAUDE.md << CEOF
# CLAUDE.md — $PROJECT_NAME
> 프로젝트별 규칙. 글로벌 규칙(~/.claude/CLAUDE.md)과 함께 적용됨.

## 프로젝트 정보
- 이름: $PROJECT_NAME
- 타입: $PROJECT_TYPE
- 보안 등급: 🟢 일반 (필요 시 🟡/🔴로 변경)

## 환경
- .env.local 확인 후 작업 시작
- 배포 URL 수정 금지 (명시적 요청 없는 한)

## 검증 명령
\`\`\`bash
npx tsc --noEmit && npx eslint . --max-warnings 0 && npm run build
\`\`\`

## Known Gotchas
<!-- 프로젝트에서 발견된 함정을 여기에 누적 기록 -->

CEOF
echo "  ✓ CLAUDE.md (프로젝트 규칙)"
else
echo "  ⏭ CLAUDE.md 이미 존재 — 건너뜀"
fi

# ── 4. PROGRESS.md ──
if [ ! -f PROGRESS.md ]; then
cat > PROGRESS.md << PEOF
# PROGRESS.md — $PROJECT_NAME
> 최종 업데이트: $(date '+%Y-%m-%d')

## 현재 위치
- Epic: 미정
- Task: 미정
- 상태: 프로젝트 초기화 완료

## 완료된 작업
- [x] 프로젝트 초기화 (.claude/ 세팅)

## 다음 할 일
- [ ] PRD 작성 또는 로드
- [ ] Epic/Task 분해

## 차단 요소
없음

PEOF
echo "  ✓ PROGRESS.md (진행 추적)"
else
echo "  ⏭ PROGRESS.md 이미 존재 — 건너뜀"
fi

# ── 5. docs/learnings.md ──
if [ ! -f docs/learnings.md ]; then
cat > docs/learnings.md << LEOF
# learnings.md — $PROJECT_NAME
> 프로젝트에서 얻은 교훈. 세션 시작 시 자동 로드.

<!-- 형식: 증상 → 원인 → 해결 → 규칙 -->

LEOF
echo "  ✓ docs/learnings.md (교훈 기록)"
else
echo "  ⏭ docs/learnings.md 이미 존재 — 건너뜀"
fi

# ── 6. 서브에이전트 ──
if [ ! -f .claude/agents/code-reviewer.md ]; then
cat > .claude/agents/code-reviewer.md << 'AEOF'
---
name: code-reviewer
description: 코드 품질 독립 검토자
tools: Read, Grep, Glob
model: sonnet
---
독립적 코드 리뷰어로서 검토:
- TypeScript strict 준수 (any 타입 = 즉시 플래그)
- SRP 위반 (함수가 2가지 이상 하면 분리 제안)
- 에러 처리 완전성 (catch에서 console.log만 있으면 플래그)
- 누락된 엣지 케이스
- 성능 이슈 (N+1, 불필요한 재렌더링)
AEOF
echo "  ✓ .claude/agents/code-reviewer.md"
fi

if [ ! -f .claude/agents/security-reviewer.md ]; then
cat > .claude/agents/security-reviewer.md << 'AEOF'
---
name: security-reviewer
description: 보안 취약점 검토
tools: Read, Grep, Glob, Bash
model: sonnet
---
시니어 보안 엔지니어로서 검토:
- 인젝션 (SQL, XSS, 커맨드 인젝션)
- 인증/인가 결함
- 하드코딩된 시크릿
- 안전하지 않은 데이터 처리
- OWASP Top 10
구체적 라인 번호와 수정 제안을 제공.
AEOF
echo "  ✓ .claude/agents/security-reviewer.md"
fi

# ── 7. gstack 복사 (설치되어 있으면) ──
if [ -d ~/.claude/skills/gstack ] && [ ! -d .claude/skills/gstack ]; then
  cp -Rf ~/.claude/skills/gstack .claude/skills/gstack
  rm -rf .claude/skills/gstack/.git
  echo "  ✓ gstack 프로젝트에 복사됨"
else
  echo "  ⏭ gstack 이미 있거나 글로벌 미설치 — 건너뜀"
fi

# ── 8. git 초기화 (필요 시) ──
if [ ! -d .git ]; then
  git init
  echo "  ✓ git 초기화"
fi

# ── 9. .gitignore에 추가 ──
if [ -f .gitignore ]; then
  grep -q "settings.local.json" .gitignore || echo ".claude/settings.local.json" >> .gitignore
else
  echo ".claude/settings.local.json" > .gitignore
fi
echo "  ✓ .gitignore 업데이트"

echo ""
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "✅ 프로젝트 초기화 완료!"
echo ""
echo "📁 생성된 구조:"
echo "  .claude/"
echo "    ├── settings.json      (프로젝트별 Hook)"
echo "    ├── agents/"
echo "    │   ├── code-reviewer.md"
echo "    │   └── security-reviewer.md"
echo "    └── skills/            (gstack 등)"
echo "  CLAUDE.md                (프로젝트 규칙)"
echo "  PROGRESS.md              (진행 추적)"
echo "  docs/learnings.md        (교훈 기록)"
echo ""
echo "👉 다음: Claude Code에서 /start 실행"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

INITEOF
chmod +x ~/.claude/init-project.sh
echo "✅ 초기화 스크립트 생성 완료: ~/.claude/init-project.sh"
```

### 2단계: 사용하기

```bash
# 새 Next.js 프로젝트 시작할 때
cd ~/project/새프로젝트
~/.claude/init-project.sh nextjs

# n8n 자동화 프로젝트
cd ~/project/새자동화
~/.claude/init-project.sh n8n

# 문서/기획 프로젝트
cd ~/project/새기획
~/.claude/init-project.sh docs
```

## 방법 2: Claude Code 스킬로 자동화 (더 스마트)

Claude Code 안에서 `/init` 으로 실행되는 스킬을 만듭니다.

### 스킬 생성

```bash
mkdir -p ~/.claude/skills/init
```

아래 내용을 `~/.claude/skills/init/SKILL.md`로 저장:

```markdown
---
name: init
description: >
  새 프로젝트 초기 세팅. .claude/ 폴더, CLAUDE.md, PROGRESS.md, 서브에이전트,
  Hook을 자동 생성한다.
  트리거: "프로젝트 세팅", "init", "초기화", 새 프로젝트 시작 시.
allowed-tools: Read, Write, Bash, Glob
---

# 프로젝트 초기화

## 실행 순서

### 1. 프로젝트 분석
- package.json 존재 여부 확인 → TypeScript 프로젝트인지 판단
- 이미 .claude/ 폴더가 있는지 확인 → 있으면 덮어쓰지 않음
- 보안 등급 판단: "돈/신분/법" 기준으로 Jayden에게 질문

### 2. Jayden에게 질문 (필수)
```
프로젝트 초기화를 시작합니다. 몇 가지 확인이 필요합니다:

1. 프로젝트 타입은? (Next.js / n8n / 기타)
2. 보안 등급은?
   - 🔴 결제/인증/개인정보 관련
   - 🟡 부분적 민감 데이터
   - 🟢 일반 콘텐츠/분석
3. PRD가 이미 있나요? (파일 경로 또는 "없음")
```

### 3. 파일 생성
Jayden 답변 후 아래를 생성:

```bash
~/.claude/init-project.sh [타입]
```

또는 스크립트가 없으면 직접:
- `.claude/settings.json` (타입에 맞는 Hook)
- `CLAUDE.md` (보안 등급 반영)
- `PROGRESS.md`
- `docs/learnings.md`
- `.claude/agents/code-reviewer.md`
- `.claude/agents/security-reviewer.md`

### 4. PRD 연결
- PRD가 있으면 → `docs/PRD.md`로 복사 또는 심링크
- PRD가 없으면 → "PRD Workshop에서 먼저 작성하세요" 안내

### 5. 초기화 보고
```
✅ 프로젝트 초기화 완료
━━━━━━━━━━━━━━━━━━━━━━━━━━
📁 프로젝트: [이름]
🏷️ 타입: [Next.js / n8n / 기타]
🔒 보안: [🔴/🟡/🟢]
📄 생성된 파일: [N]개
━━━━━━━━━━━━━━━━━━━━━━━━━━
👉 다음: /start 로 세션 시작
```

## 주의사항
- 이미 존재하는 파일은 절대 덮어쓰지 않음
- 보안 🔴 프로젝트는 CLAUDE.md에 "n8n/자동화 금지" 규칙 자동 추가
- gstack이 글로벌에 설치되어 있으면 프로젝트에 자동 복사
```

## 방법 3: 앨리어스 (가장 간편)

`.zshrc`에 단축 명령 추가:

```bash
echo '
# Claude Code 프로젝트 초기화
alias cc-init="~/.claude/init-project.sh"
alias cc-init-next="~/.claude/init-project.sh nextjs"
alias cc-init-n8n="~/.claude/init-project.sh n8n"
alias cc-init-docs="~/.claude/init-project.sh docs"
' >> ~/.zshrc
source ~/.zshrc
```

이후 사용:
```bash
cd ~/project/새프로젝트
cc-init-next    # 끝!
```

  /start                    ← 세션 시작 보고
  → Jayden 승인
  /spec2task docs/PRD.md    ← Task 분해 (PRD가 있으면)
  → 구현 시작
```

## 생성되는 전체 구조

```
프로젝트루트/
├── .claude/
│   ├── settings.json           ← 프로젝트별 Hook (tsc + eslint)
│   ├── agents/
│   │   ├── code-reviewer.md    ← 코드 리뷰 서브에이전트
│   │   └── security-reviewer.md ← 보안 리뷰 서브에이전트
│   └── skills/
│       └── gstack/             ← gstack 복사 (선택)
├── docs/
│   ├── PRD.md                  ← PRD (별도 작성)
│   └── learnings.md            ← 교훈 기록
├── CLAUDE.md                   ← 프로젝트 규칙
├── PROGRESS.md                 ← 진행 추적
└── .gitignore                  ← settings.local.json 제외
```

## 빠른 실행 (지금 바로)

1단계만 터미널에 복사-붙여넣기하면 스크립트가 생성됩니다.
